const DAY = 86400000;
function sleepSig(ms: number, sig?: AbortSignal | null, hang = false) {
  return new Promise<void>((resolve, reject) => {
    const t = hang ? undefined : setTimeout(resolve, ms);
    sig?.addEventListener("abort", () => { if (t) clearTimeout(t); reject(sig.reason ?? new Error("aborted")); });
  });
}
function runQ(q: { query: string; params: any[] }) {
  const s = q.query;
  if (/pg_advisory_xact_lock/.test(s)) return res(["x"], [[1]]);
  if (/^\s*delete from/.test(s)) { const ks = arr(q.params[0]); rows = rows.filter((r) => !(ks.includes(r.key) && r.ws > now - DAY)); return { command: "DELETE", rowCount: 1, fields: [], rows: [] }; }
  if (/^\s*select 1 from/.test(s)) { const ks = arr(q.params[0]); const h = rows.some((r) => ks.includes(r.key) && r.ws > now - DAY); return res(["c"], h ? [[1]] : []); }
  if (/with seen as/.test(s)) {
    const keys = arr(q.params[0]); const emailKey = String(q.params[2]); const counted = arr(q.params[3]);
    const seen = keys.filter((k) => rows.some((r) => r.key === k && r.ws > now - DAY));
    const out: unknown[][] = seen.map((k) => ["seen", k, 0]);
    const rpKey = String(q.params[4]);
    const emailSeen = seen.includes(emailKey);
    for (const k of keys) if (!seen.includes(k) && (emailSeen ? k === rpKey : k !== rpKey)) rows.push({ key: k, ws: now, count: 1 }); // 1fba3ec
    if (!seen.includes(emailKey)) {
      const hr = Math.floor(now / 3600000) * 3600000;
      for (const k of counted) { let r = rows.find((x) => x.key === k && x.ws === hr); if (!r) { r = { key: k, ws: hr, count: 0 }; rows.push(r); } r.count++; out.push(["ip", k, r.count]); }
    }
    return res(["kind", "key", "n"], out);
  }
  if (/^\s*insert into s1_rate_limits/.test(s)) { // hit()
    const win = Number(q.params[1]); const ws = Math.floor(now / 1000 / win) * win * 1000;
    let r = rows.find((x) => x.key === q.params[0] && x.ws === ws); if (!r) { r = { key: q.params[0], ws, count: 0 }; rows.push(r); } r.count++;
    return { command: "INSERT", rowCount: 1, fields: [{ name: "count", dataTypeID: 23 }], rows: [[r.count]] };
  }
  return res(["x"], []);
}
